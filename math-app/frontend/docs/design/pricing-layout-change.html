What changes in the Dart code

One file: lib/screens/pricing_screen.dart — the 3 tier cards (lines 50–54) are wrapped in a LayoutBuilder that switches between Row and Column based on constraints.maxWidth >= 900.

No data, no l10n, no business logic change. Pure layout.

Approx diff:

_buildTierRow(BuildContext context, AppLocalizations? l10n, PricingConfig pricing, bool isEn) {
  return LayoutBuilder(
    builder: (context, constraints) {
      final wide = constraints.maxWidth >= 900;
      final free = pricing.tierById('free')!;
      final premium = pricing.tierById('premium_monthly')!;
      final yearly = pricing.tierById('premium_yearly')!;
      if (wide) {
        return Row(
          crossAxisAlignment: CrossAxisAlignment.stretch,
          children: [
            Expanded(child: _buildTierCard(context, l10n, free, isEn: isEn, currentBadge: true)),
            const SizedBox(width: 16),
            Expanded(child: _buildTierCard(context, l10n, premium, isEn: isEn, popular: true, currentBadge: false)),
            const SizedBox(width: 16),
            Expanded(child: _buildTierCard(context, l10n, yearly, isEn: isEn, currentBadge: false)),
          ],
        );
      }
      return Column(
        children: [
          _buildTierCard(context, l10n, free, isEn: isEn, currentBadge: true),
          const SizedBox(height: 16),
          _buildTierCard(context, l10n, premium, isEn: isEn, popular: true, currentBadge: false),
          const SizedBox(height: 16),
          _buildTierCard(context, l10n, yearly, isEn: isEn, currentBadge: false),
        ],
      );
    },
  );
}
